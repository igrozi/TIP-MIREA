#include <iostream>
using namespace std;

int main() {
    const int sizeA = 4;
    const int sizeB = 4;
  
    int A[sizeA] = {1, 4, 7, 10};
    int B[sizeB] = {2, 3, 8, 12};

    int i = 0;
    int j = 0;
  
    while (i < sizeA && j < sizeB) {
        if (A[i] < B[j]) {
            cout << A[i] << " ";
            i++;
        }
        else {
            cout << B[j] << " ";
            j++;
        }
    }
    return 0;
}
